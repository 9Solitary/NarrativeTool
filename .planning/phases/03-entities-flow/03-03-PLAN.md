    DOM augmentation 逻辑（在 main.js 中实现，以下是伪代码）：

    ```javascript
    // 在 main.js 的 onload() 中：
    this._setupCanvasNodeTypeObserver();

    // 方法定义：
    _setupCanvasNodeTypeObserver() {
        // 使用 MutationObserver 监听 .canvas-wrapper 元素中的变化
        const observer = new MutationObserver((mutations) => {
            for (const mutation of mutations) {
                for (const node of mutation.addedNodes) {
                    if (node.nodeType === 1) { // Element node
                        this._annotateCanvasNodes(node);
                    }
                }
            }
        });

        // 初始扫描：如果 Canvas 已经打开
        const existingCanvas = document.querySelector('.canvas-wrapper');
        if (existingCanvas) {
            this._annotateCanvasNodes(existingCanvas);
        }

        // 监听 workspace layout-change 事件以检测新 Canvas 打开
        this.registerEvent(
            this.app.workspace.on('layout-change', () => {
                const wrappers = document.querySelectorAll('.canvas-wrapper');
                wrappers.forEach(w => {
                    observer.observe(w, { childList: true, subtree: true });
                    this._annotateCanvasNodes(w);
                });
            })
        );
    }

    _annotateCanvasNodes(container) {
        // 获取当前 canvas file path 以读取 JSON
        const canvasView = this.app.workspace.getActiveViewOfType(/* Canvas view type */);
        if (!canvasView || !canvasView.file) return;

        const canvasNodes = container.querySelectorAll('.canvas-node');
        canvasNodes.forEach(nodeEl => {
            // Skip if already annotated
            if (nodeEl.hasAttribute('data-nt-type')) return;

            // 检查 node 内容以确定类型
            const nodeContent = nodeEl.querySelector('.canvas-node-content');
            if (!nodeContent) return;

            const label = nodeContent.textContent || '';
            // 检测 file extension 标记
            if (label.endsWith('.ncanvas')) {
                nodeEl.setAttribute('data-nt-type', 'dialogue');
            } else if (label.includes('character') || label.endsWith('.md')) {
                // 更精确的类型检测：需要与 .canvas JSON 交叉引用
                // 简化方案：检查 label 中是否包含类型关键词
                // 增强方案：读取 .canvas JSON 解析 file path，再检查 .md frontmatter
                // Phase 3 实现简化方案；Phase 4 增强为读取 frontmatter
                // 注：对于 .md files，无法仅从 label 判断是 character/location/quest/item
                // 简化方案：将所有 .md file nodes 标记为 'entity' 并使用中性样式
                nodeEl.setAttribute('data-nt-type', 'entity');
            }
        });
    }
    ```

    **⚠ 实现简化决策（根据 03-RESEARCH.md Open Question 2 "Canvas DOM structure stability"）：**

                    const labelEl = nodeEl.querySelector('.canvas-node-label');
                    if (!labelEl) return;

                    const label = labelEl.textContent || '';
                    if (label.endsWith('.ncanvas')) {
                        nodeEl.setAttribute('data-nt-type', 'dialogue');
                    } else if (label.endsWith('.md')) {
                        nodeEl.setAttribute('data-nt-type', 'entity');
                    }
                });

    ```javascript
    function promptForInput(app, title, placeholder) {
        return new Promise((resolve) => {
            const modal = new (require('obsidian').Modal)(app);
            modal.titleEl.setText(title);
            const input = modal.contentEl.createEl('input', {
                type: 'text',
                placeholder: placeholder
            });
            input.style.width = '100%';
            input.addEventListener('keydown', (e) => {
                if (e.key === 'Enter') {
                    modal.close();
                    resolve(input.value);
                }
            });
            modal.onClose = () => resolve(input.value); // 关闭时也 resolve（可能为空）
            modal.open();
        });
    }
    ```

    注意：此 prompt 函数使用回调，需要正确的 async/await 封装。
